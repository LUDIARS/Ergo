# Fake host for test_m0_invariance.cmake (acceptance C-5). Creates a pictor
# target the way existing hosts do, includes the package hub, then proves the
# hub changed neither the target set nor any compile definition.
cmake_minimum_required(VERSION 3.20)
project(ergo_pkg_m0_host NONE)

# Snapshot of every build-system target with its type and compile
# definitions, plus the directory-level definitions.
function(m0_snapshot label)
    get_property(_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    list(SORT _targets)
    set(_lines "")
    foreach(_t IN LISTS _targets)
        get_target_property(_type ${_t} TYPE)
        get_target_property(_idefs ${_t} INTERFACE_COMPILE_DEFINITIONS)
        set(_defs "")
        if(NOT _type STREQUAL "INTERFACE_LIBRARY")
            get_target_property(_defs ${_t} COMPILE_DEFINITIONS)
        endif()
        list(APPEND _lines "${_t}|${_type}|${_defs}|${_idefs}")
    endforeach()
    get_property(_dir_defs DIRECTORY PROPERTY COMPILE_DEFINITIONS)
    list(APPEND _lines "<dir>|${_dir_defs}")
    set_property(GLOBAL PROPERTY M0_SNAPSHOT_${label} "${_lines}")
endfunction()

function(m0_expect_equal a b)
    get_property(_a GLOBAL PROPERTY M0_SNAPSHOT_${a})
    get_property(_b GLOBAL PROPERTY M0_SNAPSHOT_${b})
    if(NOT _a STREQUAL _b)
        message(FATAL_ERROR "C-5: targets / compile definitions changed between ${a} and ${b}\n  ${a}: ${_a}\n  ${b}: ${_b}")
    endif()
endfunction()

add_library(pictor INTERFACE)
target_compile_definitions(pictor INTERFACE PICTOR_HAS_VULKAN=1)
add_library(host_support INTERFACE)
target_compile_definitions(host_support INTERFACE HOST_SUPPORT=1)
set_property(DIRECTORY APPEND PROPERTY COMPILE_DEFINITIONS HOST_DIR_DEF=1)

m0_snapshot(before_hub)
include("${ERGO_SOURCE_DIR}/cmake/packages/ErgoPackages.cmake")
m0_snapshot(after_hub)
m0_expect_equal(before_hub after_hub)

# Stand-in for the module sections that follow the hub in Ergo.
add_library(ergo_render INTERFACE)
if(NOT DEFINED FIXTURE_REAL_PATH)
    set(FIXTURE_REAL_PATH OFF)
endif()
set(ERGO_RENDER_BACKEND_ENABLED ${FIXTURE_REAL_PATH})
set(ERGO_RENDER_BACKEND_REASON "fixture: real path ${FIXTURE_REAL_PATH}")
m0_snapshot(after_modules)

# Runs after the hub's own deferred finalize (registered earlier).
function(m0_check_end)
    m0_snapshot(after_finalize)
    m0_expect_equal(after_modules after_finalize)

    set(_resolved "${CMAKE_BINARY_DIR}/ergo-packages.resolved.json")
    if(NOT EXISTS "${_resolved}")
        message(FATAL_ERROR "C-5: ${_resolved} was not written")
    endif()
    file(READ "${_resolved}" _json)
    string(JSON _mode GET "${_json}" mode)
    if(NOT _mode STREQUAL FIXTURE_EXPECT_MODE)
        message(FATAL_ERROR "C-5: mode '${_mode}', expected '${FIXTURE_EXPECT_MODE}'")
    endif()
    string(JSON _pictor GET "${_json}" packages pictor state)
    string(JSON _tela GET "${_json}" packages tela state)
    if(NOT _pictor STREQUAL "host-unmanaged" OR NOT _tela STREQUAL "absent")
        message(FATAL_ERROR "C-5: states pictor='${_pictor}' tela='${_tela}'")
    endif()
    string(JSON _render GET "${_json}" modules ergo_render enabled)
    if(NOT _render)
        message(FATAL_ERROR "C-5: ergo_render should be recorded as enabled")
    endif()
    if(EXISTS "${CMAKE_BINARY_DIR}/.cmake/api/v1/query/client-ergo/query.json")
        file(WRITE "${CMAKE_BINARY_DIR}/m0_fixture_ok.txt" "ok\n")
    else()
        message(FATAL_ERROR "C-5: File API query was not installed")
    endif()
endfunction()
cmake_language(DEFER CALL m0_check_end)
